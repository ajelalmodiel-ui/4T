local function startAutoLook()
    if autoLookConn then return end
    local char = player.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end
    local hum2 = char:FindFirstChildOfClass("Humanoid"); if not hum2 then return end
    hum2.AutoRotate = false
    -- Attachment + AlignOrientation so it works even while velocity is set
    if autoLookAttachment then autoLookAttachment:Destroy() end
    if autoLookAlign then autoLookAlign:Destroy() end
    autoLookAttachment = Instance.new("Attachment", root)
    autoLookAlign = Instance.new("AlignOrientation", root)
    autoLookAlign.Attachment0 = autoLookAttachment
    autoLookAlign.Mode = Enum.OrientationAlignmentMode.OneAttachment
    autoLookAlign.MaxTorque = Vector3.new(0, math.huge, 0)
    autoLookAlign.Responsiveness = 200
    autoLookAlign.RigidityEnabled = false
    autoLookConn = RunService.RenderStepped:Connect(function()
        if patrolMode == "none" then return end
        local c2 = player.Character; if not c2 then return end
        local r2 = c2:FindFirstChild("HumanoidRootPart"); if not r2 then return end
        if not autoLookAlign or not autoLookAlign.Parent then return end
        -- Find nearest steal target position (enemy brainrot podium)
        local bestPos = nil
        local bestDist = math.huge
        pcall(function()
            local plots = workspace:FindFirstChild("Plots"); if not plots then return end
            for _, plot in ipairs(plots:GetChildren()) do
                if not isMyPlotByName(plot.Name) then
                    local podiums = plot:FindFirstChild("AnimalPodiums")
                    if podiums then
                    for _, pod in ipairs(podiums:GetChildren()) do
                        local base = pod:FindFirstChild("Base")
                        local spawn = base and base:FindFirstChild("Spawn")
                        if spawn then
                            local d = (spawn.Position - r2.Position).Magnitude
                            if d < bestDist then bestDist = d; bestPos = spawn.Position end
                        end
                    end
                    end -- close podiums if
                end
            end
        end)
        if bestPos then
            -- Rotate body flat toward the brainrot
            local lookPos = Vector3.new(bestPos.X, r2.Position.Y, bestPos.Z)
            autoLookAlign.CFrame = CFrame.lookAt(r2.Position, lookPos)
            -- Snap camera to also face the brainrot
            pcall(function()
                local cam = workspace.CurrentCamera
                if cam then
                    local camPos = cam.CFrame.Position
                    local camLook = Vector3.new(bestPos.X, camPos.Y, bestPos.Z)
                    cam.CFrame = CFrame.lookAt(camPos, camLook)
                end
            end)
        end
    end)
end

-- Pure Lua JSON (no HttpService needed)
local _json = {}
do
    local function enc(v)
        local t=type(v)
        if t=="nil" then return "null"
        elseif t=="boolean" then return tostring(v)
        elseif t=="number" then if v~=v then return "null" end; return tostring(v)
        elseif t=="string" then
            return '"'..v:gsub('\\','\\\\'):gsub('"','\\"'):gsub('\n','\\n'):gsub('\r','\\r'):gsub('\t','\\t')..'"'
        elseif t=="table" then
            if #v>0 then
                local p={}; for _,val in ipairs(v) do p[#p+1]=enc(val) end
                return "["..table.concat(p,",").."]"
            else
                local p={}; for k,val in pairs(v) do
                    if type(k)=="string" or type(k)=="number" then
                        p[#p+1]='"'..tostring(k)..'":'..enc(val)
                    end
                end
                return "{"..table.concat(p,",").."}"
            end
        end; return "null"
    end
    function _json.encode(v) return enc(v) end
    local dv
    local function sw(s,i) while i<=#s and s:sub(i,i):match("[ \t\n\r]") do i=i+1 end; return i end
    local function ds(s,i)
        i=i+1; local r={}
        while i<=#s do
            local ch=s:sub(i,i)
            if ch=='"' then return table.concat(r),i+1 end
            if ch=='\\' then i=i+1; local e=s:sub(i,i)
                if e=='n' then r[#r+1]='\n' elseif e=='t' then r[#r+1]='\t' else r[#r+1]=e end
            else r[#r+1]=ch end; i=i+1
        end; return table.concat(r),i
    end
    local function da(s,i) local a={}; i=sw(s,i)
        if s:sub(i,i)=="]" then return a,i+1 end
        while true do local v2; v2,i=dv(s,i); a[#a+1]=v2; i=sw(s,i)
            if s:sub(i,i)=="]" then return a,i+1 end; i=sw(s,i+1) end
    end
    local function do2(s,i) local o={}; i=sw(s,i)
        if s:sub(i,i)=="}" then return o,i+1 end
        while true do i=sw(s,i); local k; k,i=ds(s,i)
            i=sw(s,i)+1; i=sw(s,i); local v2; v2,i=dv(s,i); o[k]=v2; i=sw(s,i)
            if s:sub(i,i)=="}" then return o,i+1 end; i=sw(s,i+1) end
    end
    dv=function(s,i) i=sw(s,i); local ch=s:sub(i,i)
        if ch=='"' then return ds(s,i)
        elseif ch=='[' then return da(s,i+1)
        elseif ch=='{' then return do2(s,i+1)
        elseif s:sub(i,i+3)=="true"  then return true, i+4
        elseif s:sub(i,i+4)=="false" then return false,i+5
        elseif s:sub(i,i+3)=="null"  then return nil,  i+4
        else local n=s:match("^-?%d+%.?%d*",i); if n then return tonumber(n),i+#n end end
        return nil,i+1
    end
    function _json.decode(s)
        local ok,v=pcall(function() return(dv(s,1)) end); return ok and v or nil
    end
end

local function saveConfig()
    local ok, err = pcall(function()
        -- Numbers
        local data = {
            NORMAL_SPEED    = NORMAL_SPEED,
            SLOW_SPEED      = SLOW_SPEED,
            STEAL_RADIUS    = STEAL_RADIUS,
            STEAL_DURATION  = STEAL_DURATION,
            GALAXY_GRAVITY  = GALAXY_GRAVITY,
            GALAXY_HOP      = GALAXY_HOP,
            SPIN_SPEED      = SPIN_SPEED,
            fovValue        = fovValue,
            AIMBOT_SPEED    = AIMBOT_SPEED,
            BAT_ENGAGE_RANGE= BAT_ENGAGE_RANGE,
            grabInterval    = grabInterval,
        }
        -- Toggle states (features on/off)
        for k, v in pairs(toggleStates) do
            data["TOGGLE_"..k] = v.state
        end
        -- Keybinds
        for k, v in pairs(Keybinds) do
            if v then data["KEY_"..k] = v.Name end
        end
        -- Save UI positions
        if MainFrame then
            local p=MainFrame.Position
            data.mainFramePos={xs=p.X.Scale,x=p.X.Offset,ys=p.Y.Scale,y=p.Y.Offset}
        end
        if TitleBar then
            local p=TitleBar.Position
            data.titleBarPos={xs=p.X.Scale,x=p.X.Offset,ys=p.Y.Scale,y=p.Y.Offset}
        end
        -- Save mobile button positions
        data.mobileBtnPositions = {}
        for i, btn in ipairs(mobileButtons) do
            local p = btn.Position
            data.mobileBtnPositions[i] = {xs=p.X.Scale,x=p.X.Offset,ys=p.Y.Scale,y=p.Y.Offset}
        end
        -- Save waypoints
        local function serWp(list)
            local t={}; for i,v in ipairs(list) do t[i]={x=v.X,y=v.Y,z=v.Z} end; return t
        end
        data.leftWaypoints  = serWp(leftWaypoints)
        data.rightWaypoints = serWp(rightWaypoints)
        writefile(CONFIG_KEY..".json", _json.encode(data))
    end)
    return ok
end

local function loadConfig()
    pcall(function()
        if not (readfile and isfile and isfile(CONFIG_KEY..".json")) then return end
        local ok, data = pcall(function()
            return _json.decode(readfile(CONFIG_KEY..".json"))
        end)
        if not (ok and data) then return end
        -- Numbers
        if data.NORMAL_SPEED     then NORMAL_SPEED     = data.NORMAL_SPEED     end
        if data.SLOW_SPEED       then SLOW_SPEED       = data.SLOW_SPEED       end
        if data.STEAL_RADIUS     then STEAL_RADIUS     = data.STEAL_RADIUS     end
        if data.STEAL_DURATION   then STEAL_DURATION   = data.STEAL_DURATION   end
        if data.GALAXY_GRAVITY   then GALAXY_GRAVITY   = data.GALAXY_GRAVITY   end
        if data.GALAXY_HOP       then GALAXY_HOP       = data.GALAXY_HOP       end
        if data.SPIN_SPEED       then SPIN_SPEED       = data.SPIN_SPEED       end
        if data.fovValue         then fovValue         = data.fovValue         end
        if data.AIMBOT_SPEED     then AIMBOT_SPEED     = data.AIMBOT_SPEED     end
        if data.BAT_ENGAGE_RANGE then BAT_ENGAGE_RANGE = data.BAT_ENGAGE_RANGE end
        if data.grabInterval and data.grabInterval >= 0.05 then
            grabInterval = data.grabInterval
        end
        -- Keybinds
        for k, _ in pairs(Keybinds) do
            if data["KEY_"..k] then
                pcall(function() Keybinds[k] = Enum.KeyCode[data["KEY_"..k]] end)
            end
        end
        -- Toggle states (applied after UI exists via SavedToggleStates)
        SavedToggleStates = {}
        for k, v in pairs(data) do
            if k:sub(1,7) == "TOGGLE_" then
                SavedToggleStates[k:sub(8)] = v
            end
        end
        -- UI positions (applied after GUI exists)
        if data.mainFramePos then
            local p=data.mainFramePos
            task.delay(0.5, function()
                if MainFrame then MainFrame.Position=UDim2.new(p.xs or 0.5,p.x,p.ys or 0.5,p.y) end
            end)
        end
        if data.titleBarPos then
            local p=data.titleBarPos
            task.delay(0.5, function()
                if TitleBar then TitleBar.Position=UDim2.new(p.xs or 0.5,p.x,p.ys or 0,p.y) end
            end)
        end
        if data.mobileBtnPositions then
            task.delay(0.6, function()
                for i, p in ipairs(data.mobileBtnPositions) do
                    if mobileButtons[i] then
                        mobileButtons[i].Position=UDim2.new(p.xs or 1,p.x,p.ys or 0.5,p.y)
                    end
                end
            end)
        end
        -- Restore waypoints
        if data.leftWaypoints then
            for i,v in ipairs(data.leftWaypoints) do
                leftWaypoints[i] = Vector3.new(v.x, v.y, v.z)
            end
        end
        if data.rightWaypoints then
            for i,v in ipairs(data.rightWaypoints) do
                rightWaypoints[i] = Vector3.new(v.x, v.y, v.z)
            end
        end
    end)
end

    local plots=workspace:FindFirstChild("Plots") if not plots then return "Left" end

                local center=plot:FindFirstChildWhichIsA("BasePart")

        end
    end

local function hookTPCharacter(char)
    if tpStateConn then tpStateConn:Disconnect() tpStateConn=nil end
    if tpChildConn then tpChildConn:Disconnect() tpChildConn=nil end
    if tpChildRemConn then tpChildRemConn:Disconnect() tpChildRemConn=nil end
    local hum=char:WaitForChild("Humanoid")
    local spawnTime=tick()
    local function isSpawnGrace() return (tick()-spawnTime)<2 end
    tpStateConn=hum.StateChanged:Connect(function(_,newState)
        if not RagdollTPEnabled then return end
        if isSpawnGrace() then tpWasRagdolled=false return end
        if TP_RAGDOLL_STATES[newState] then
            if not tpWasRagdolled then tpWasRagdolled=true task.defer(doRagdollTP) end
        else tpWasRagdolled=false end
    end)
    tpChildConn=char.ChildAdded:Connect(function(child)
        if not RagdollTPEnabled then return end
        if isSpawnGrace() then return end
        if isMedusaTPChild(child.Name) then
            if not tpMedusaActive then tpMedusaActive=true task.defer(doMedusaTP) end
            return
        end
        if child.Name=="Ragdoll" or child.Name=="IsRagdoll" then
            if not tpWasRagdolled then tpWasRagdolled=true task.defer(doRagdollTP) end
        end
    end)
    tpChildRemConn=char.ChildRemoved:Connect(function(child)
        if child.Name=="Ragdoll" or child.Name=="IsRagdoll" then tpWasRagdolled=false end
        if isMedusaTPChild(child.Name) then tpMedusaActive=false end
    end)
end

    if tpChildRemConn then tpChildRemConn:Disconnect() tpChildRemConn=nil end
    tpWasRagdolled=false tpMedusaActive=false
end

-- AUTO STEAL (Void Hub method — uses getconnections for precise trigger)
local isStealing=false
local StealData={}
local autoStealConn=nil
local stealStartTime=nil
local progressConnection=nil

local function isMyPlotByName(pn)
    local plots=workspace:FindFirstChild("Plots"); if not plots then return false end
    local plot=plots:FindFirstChild(pn); if not plot then return false end
    local sign=plot:FindFirstChild("PlotSign"); if sign then
        local yb=sign:FindFirstChild("YourBase")
        if yb and yb:IsA("BillboardGui") then return yb.Enabled==true end
    end
    return false
end

local function findNearestSteal()
    local c=player.Character; local root=c and c:FindFirstChild("HumanoidRootPart"); if not root then return nil end
    local plots=workspace:FindFirstChild("Plots"); if not plots then return nil end
    local np,nd,nn=nil,math.huge,nil
    for _,plot in ipairs(plots:GetChildren()) do
        if not isMyPlotByName(plot.Name) then
        local podiums=plot:FindFirstChild("AnimalPodiums"); if podiums then
        for _,pod in ipairs(podiums:GetChildren()) do
            pcall(function()
                local base=pod:FindFirstChild("Base")
                local spawn=base and base:FindFirstChild("Spawn")
                if spawn then
                    local dist=(spawn.Position-root.Position).Magnitude
                    if dist<nd and dist<=STEAL_RADIUS then
                        local att=spawn:FindFirstChild("PromptAttachment")
                        if att then
                            for _,ch in ipairs(att:GetChildren()) do
                                if ch:IsA("ProximityPrompt") then np,nd,nn=ch,dist,pod.Name; break end
                            end
                        end
                    end
                end
            end)
        end
        end end
    end
    return np,nd,nn
end

local function findClosestStealAny()
    -- kept for the UI bar progress display compatibility
    local _,nd,_=findNearestSteal()
    return nil, nd or math.huge, STEAL_RADIUS
end

local function ResetStealBar()
    if stealBarFill then stealBarFill.Size=UDim2.new(0,0,1,0) end
    if stealStatusLbl then stealStatusLbl.TextColor3=Color3.fromRGB(255,255,255) end
end

local function executeSteal(prompt, name)
    if isStealing then return end
    if not StealData[prompt] then
        StealData[prompt]={hold={},trigger={},ready=true}
        pcall(function()
            if getconnections then
                for _,c2 in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                    if c2.Function then table.insert(StealData[prompt].hold,c2.Function) end
                end
                for _,c2 in ipairs(getconnections(prompt.Triggered)) do
                    if c2.Function then table.insert(StealData[prompt].trigger,c2.Function) end
                end
            end
        end)
    end
    local data=StealData[prompt]
    -- force reset if stuck
    if not data.ready then data.ready=true; isStealing=false end
    data.ready=false; isStealing=true; stealStartTime=tick()
    if stealStatusLbl then stealStatusLbl.TextColor3=Color3.fromRGB(255,255,255) end
    -- Fast looping bar: 0→full in 0.15s, loops until steal is done
    if progressConnection then progressConnection:Disconnect(); progressConnection=nil end
    if stealBarFill then stealBarFill.Size=UDim2.new(0,0,1,0) end
    local barStart=tick()
    local BAR_LOOP_TIME=0.15 -- one full sweep in 0.15s = fast visual
    progressConnection=RunService.Heartbeat:Connect(function()
        if not stealBarFill then return end
        local elapsed=(tick()-barStart)%BAR_LOOP_TIME
        local prog=elapsed/BAR_LOOP_TIME
        stealBarFill.Size=UDim2.new(prog,0,1,0)
    end)
    task.spawn(function()
        for _,f in ipairs(data.hold) do task.spawn(f) end
        task.wait(STEAL_DURATION)
        for _,f in ipairs(data.trigger) do task.spawn(f) end
        if progressConnection then progressConnection:Disconnect(); progressConnection=nil end
        if stealBarFill then stealBarFill.Size=UDim2.new(1,0,1,0) end
        task.wait(0.03)
        ResetStealBar(); data.ready=true; isStealing=false
    end)
end

local function startAutoSteal()
    if autoStealConn then return end
    autoStealConn=RunService.Heartbeat:Connect(function()
        if not autoStealEnabled then return end
        -- force clear stuck state
        if isStealing and stealStartTime and (tick()-stealStartTime)>STEAL_DURATION*3 then
            isStealing=false
            if progressConnection then progressConnection:Disconnect(); progressConnection=nil end
            for _,d in pairs(StealData) do d.ready=true end
            ResetStealBar()
        end
        if isStealing then return end
        local p,_,n=findNearestSteal()
        if p then executeSteal(p,n) end
    end)
end

local function stopAutoSteal()
    if autoStealConn then autoStealConn:Disconnect(); autoStealConn=nil end
    isStealing=false
    if progressConnection then progressConnection:Disconnect(); progressConnection=nil end
    ResetStealBar()
end

local function startBatAimbot()
    if aimbotConnection then return end
    local c=player.Character if not c then return end
    local h=c:FindFirstChild("HumanoidRootPart") local hum=c:FindFirstChildOfClass("Humanoid")
    if not h or not hum then return end
    hum.AutoRotate=false
    local attachment=h:FindFirstChild("AimbotAttachment") or Instance.new("Attachment",h)
    attachment.Name="AimbotAttachment"
    local align=h:FindFirstChild("AimbotAlign") or Instance.new("AlignOrientation",h)
    align.Name="AimbotAlign" align.Mode=Enum.OrientationAlignmentMode.OneAttachment
    align.Attachment0=attachment align.MaxTorque=math.huge align.Responsiveness=200
    batAimbotEnabled=true
    aimbotConnection=RunService.Heartbeat:Connect(function()
        if not batAimbotEnabled then return end
        if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
        local currentHRP=player.Character.HumanoidRootPart
        local currentHum=player.Character:FindFirstChildOfClass("Humanoid")
        -- Only equip bat/medusa if player has NO tool currently equipped
        local heldTool=player.Character:FindFirstChildOfClass("Tool")
        if not heldTool then
            local bat=findBatTool()
            if bat then pcall(function() currentHum:EquipTool(bat) end) end
            heldTool=player.Character:FindFirstChildOfClass("Tool")
        end
        local targetHRP,targetChar=getBestTarget(currentHRP)
        if targetHRP and targetChar then
            aimbotHighlight.Adornee=targetChar
            local targetVelocity=targetHRP.AssemblyLinearVelocity
            local speed=targetVelocity.Magnitude
            local dynamicPredictTime=math.clamp(speed/150,0.05,0.2)
            local predictedPos=targetHRP.Position+(targetVelocity*dynamicPredictTime)
            local dirToTarget=(predictedPos-currentHRP.Position)
            local distance3D=dirToTarget.Magnitude
            local targetStandPos=predictedPos
            if distance3D>0 then targetStandPos=predictedPos-(dirToTarget.Unit*MELEE_OFFSET) end
            align.CFrame=CFrame.lookAt(currentHRP.Position,predictedPos)
            local moveDir=(targetStandPos-currentHRP.Position)
            local distToStandPos=moveDir.Magnitude
            if distToStandPos>1 then currentHRP.AssemblyLinearVelocity=moveDir.Unit*AIMBOT_SPEED
            else currentHRP.AssemblyLinearVelocity=targetVelocity end
            if distToStandPos<=BAT_ENGAGE_RANGE then
                if heldTool and heldTool.Parent==player.Character then
                    local isMedusa = heldTool.Name:lower():find("medusa")
                    if isMedusa then
                        if not medusaFired or medusaLastTarget ~= targetChar then
                            medusaFired = true
                            medusaLastTarget = targetChar
                            pcall(function() heldTool:Activate() end)
                        end
                    else
                        pcall(function() heldTool:Activate() end)
                    end
                end
            end
        else
            lockedTarget=nil medusaFired=false medusaLastTarget=nil currentHRP.AssemblyLinearVelocity=Vector3.new(0,0,0) aimbotHighlight.Adornee=nil
        end
    end)
end
local function stopBatAimbot()

    if aimbotConnection then aimbotConnection:Disconnect() aimbotConnection=nil end

    end

    lockedTarget=nil medusaFired=false medusaLastTarget=nil aimbotHighlight.Adornee=nil
end

                    pcall(function() medusa:Activate() end)

                end

-- ═══ NEW DUEL KING MOVEMENT ═══
local function getCurrentSpeed()
    if patrolMode == "right" then return currentWaypoint >= 3 and SLOW_SPEED or NORMAL_SPEED
    elseif patrolMode == "left" then return currentWaypoint >= 3 and SLOW_SPEED or NORMAL_SPEED end
    return 0
end

local function getCurrentWaypoints()
    if patrolMode == "right" then return rightWaypoints
    elseif patrolMode == "left" then return leftWaypoints end
    return {}
end

local function updateAutoPathWalking()
    local char = player.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end
    if patrolMode == "none" then return end
    local waypoints = getCurrentWaypoints()
    if not waypoints or #waypoints == 0 then stopMovement() return end
    local targetPos = waypoints[currentWaypoint]
    if not targetPos then stopMovement() return end
    local currentPos = root.Position
    local targetXZ  = Vector3.new(targetPos.X, 0, targetPos.Z)
    local currentXZ = Vector3.new(currentPos.X, 0, currentPos.Z)
    local distanceXZ = (targetXZ - currentXZ).Magnitude
    if distanceXZ > 3 then
        local moveDir = (targetXZ - currentXZ).Unit
        local spd = getCurrentSpeed()
        root.AssemblyLinearVelocity = Vector3.new(moveDir.X*spd, root.AssemblyLinearVelocity.Y, moveDir.Z*spd)
    else
        if currentWaypoint == #waypoints then
            root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
            patrolMode = "none"
            currentWaypoint = 1
            task.defer(function()
                if autoPathHeartbeat then autoPathHeartbeat:Disconnect(); autoPathHeartbeat = nil end
                updateToggle("Auto Play Left", false)
                updateToggle("Auto Play Right", false)
                setMobAutoColor(mobAutoLeftBtn, false)
                setMobAutoColor(mobAutoRightBtn, false)
            end)
        else
            currentWaypoint = currentWaypoint + 1
        end
    end
end

local function startAutoPath(mode)
    -- Mutual exclusion
    if patrolMode ~= "none" and patrolMode ~= mode then
        if autoPathHeartbeat then autoPathHeartbeat:Disconnect(); autoPathHeartbeat = nil end
        updateToggle("Auto Play Left", false)
        updateToggle("Auto Play Right", false)
    end
    patrolMode = mode
    currentWaypoint = 1
    if not autoPathHeartbeat then
        autoPathHeartbeat = RunService.Heartbeat:Connect(updateAutoPathWalking)
    end
    -- Start look-at brainrot when auto play begins
    task.defer(startAutoLook)
end

local function stopAutoPath()
    patrolMode = "none"
    currentWaypoint = 1
    if autoPathHeartbeat then
        autoPathHeartbeat:Disconnect()
        autoPathHeartbeat = nil
    end
    -- Stop look-at when auto play stops
    stopAutoLook()
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0) end
    end
end

local function stopMovement()
    stopAutoPath()
    updateToggle("Auto Play Left", false)
    updateToggle("Auto Play Right", false)
    -- mobAutoLeftBtn/Right are declared later in the do block; use pcall-safe refs
    task.defer(function()
        setMobAutoColor(mobAutoLeftBtn, false)
        setMobAutoColor(mobAutoRightBtn, false)
    end)
end

local function makeToggle(parent,name,order,callback,defaultOn)
    local row=Instance.new("Frame",parent)
    row.Size=UDim2.new(1,0,0,44) row.BackgroundColor3=CARD row.BorderSizePixel=0 row.LayoutOrder=order
    Instance.new("UICorner",row).CornerRadius=UDim.new(0,10)

    local pW,pH,dSz=46,24,18
    local track=Instance.new("Frame",row)
    track.Size=UDim2.new(0,pW,0,pH) track.Position=UDim2.new(1,-(pW+12),0.5,-pH/2)
    local initState=(SavedToggleStates[name]~=nil) and SavedToggleStates[name] or (defaultOn or false)
    track.BackgroundColor3=initState and GREEN or OFF_CLR track.BorderSizePixel=0
    Instance.new("UICorner",track).CornerRadius=UDim.new(1,0)
    local dot=Instance.new("Frame",track)
    dot.Size=UDim2.new(0,dSz,0,dSz)
    dot.Position=initState and UDim2.new(1,-dSz-3,0.5,-dSz/2) or UDim2.new(0,3,0.5,-dSz/2)
    dot.BackgroundColor3=Color3.fromRGB(255,255,255) dot.BorderSizePixel=0
    Instance.new("UICorner",dot).CornerRadius=UDim.new(1,0)
    toggleStates[name]={track=track,dot=dot,state=initState,dotSz=dSz}
    if initState and callback then task.defer(function() callback(true) end) end
    local btn=Instance.new("TextButton",row)
    btn.Size=UDim2.new(1,0,1,0) btn.BackgroundTransparency=1 btn.Text=""
    btn.MouseButton1Click:Connect(function()

        task.defer(saveConfig)
    end)

                changingKeybind = nil

                keyBtn.TextColor3 = ACCENT
                TweenService:Create(kStroke, TweenInfo.new(0.15), {Color=ACCENT}):Play()

                if kc == Enum.KeyCode.LeftShift  or kc == Enum.KeyCode.RightShift
                or kc == Enum.KeyCode.LeftAlt    or kc == Enum.KeyCode.RightAlt
                or kc == Enum.KeyCode.LeftMeta   or kc == Enum.KeyCode.RightMeta then
                    return
                end

                if kc == Enum.KeyCode.LeftControl or kc == Enum.KeyCode.RightControl then
                    Keybinds[keyName] = nil
                    keyBtn.Text = "NONE"
                else
                    Keybinds[keyName] = kc
                    keyBtn.Text = keyDisplayName(kc)
                end
                keyBtn.TextColor3 = ACCENT
                TweenService:Create(kStroke, TweenInfo.new(0.15), {Color=ACCENT}):Play()
                changingKeybind = nil
                conn:Disconnect()
                task.defer(saveConfig)
            end)
        end)

-- ── U/L LOCK + DRAGGABLE (from LexHub v22) ─────────────────────
local function makeDraggable(frame)
    local drag,di,ds,sp=false,nil,nil,nil
    frame.InputBegan:Connect(function(inp)
        if uiLocked then return end
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            drag=true; ds=inp.Position; sp=frame.Position
        end
    end)
    frame.InputChanged:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseMovement or inp.UserInputType==Enum.UserInputType.Touch then di=inp end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if inp==di and drag then local d=inp.Position-ds; frame.Position=UDim2.new(sp.X.Scale,sp.X.Offset+d.X,sp.Y.Scale,sp.Y.Offset+d.Y) end
    end)
    frame.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then drag=false end
    end)
end

    local LAGGER_CONFIG = isMobile and {
        TableIncrease = 290, Tries = 1, LoopWaitTime = 0.85
    } or {
        TableIncrease = 265, Tries = 1, LoopWaitTime = 0.05
    }

    local CUSTOM_REMOTE_PATH = "RobloxReplicatedStorage.SetPlayerBlockList"

    local function resolveRemote(path)
        if not path or path == "" then return nil end
        local obj = game
        local cleaned = path:gsub("^game%.", "")
        for segment in cleaned:gmatch("[^%.]+") do
            obj = obj and obj[segment]
        end
        return obj
    end

    local function getmaxvalue(val)
        if type(val) ~= "number" then return nil end
        return 499999 / (val + 2)
    end

    local function bomb(tableincrease, tries)
        local maintable = {}
        local spammedtable = {}
        table.insert(spammedtable, {})
        local z = spammedtable[1]
        for i = 1, tableincrease do
            local tableins = {}
            table.insert(z, tableins)
            z = tableins
        end
        local maximum = getmaxvalue(tableincrease) or 9999999
        for i = 1, maximum do
            table.insert(maintable, spammedtable)
            if i % 5000 == 0 then task.wait() end
        end
        local remote = resolveRemote(CUSTOM_REMOTE_PATH)
        if remote then
            for i = 1, tries do
                pcall(function()
                    if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
                        remote:FireServer(maintable)
                    elseif remote:IsA("RemoteFunction") then
                        remote:InvokeServer(maintable)
                    end
                end)
            end
        end
    end

    local lagEnabled = false
    local lagThread  = nil

    local function startLaggerLoop()
        while lagEnabled do
            pcall(function() game:GetService("NetworkClient"):SetOutgoingKBPSLimit(math.huge) end)
            task.spawn(function() bomb(LAGGER_CONFIG.TableIncrease, LAGGER_CONFIG.Tries) end)
            task.wait(math.max(LAGGER_CONFIG.LoopWaitTime, 0.15))
        end
    end

    local function stopLag()
        lagEnabled = false
        if lagThread then pcall(coroutine.close, lagThread); lagThread = nil end
    end

    local function startLag()
        if lagThread then return end
        lagEnabled = true
        lagThread = coroutine.create(startLaggerLoop)
